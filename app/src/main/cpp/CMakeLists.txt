cmake_minimum_required(VERSION 3.22.1)
project(openspeech-phonemes VERSION 1.52.0 LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../.. ABSOLUTE)
set(ESPEAK ${ROOT}/third_party/espeak-ng)
# Build only the vendored phonemizer library. No audio driver, asynchronous worker,
# proprietary MBROLA voice, Sonic dependency, network fetch or generated executable.
set(BUILD_SHARED_LIBS OFF)
set(HAVE_MKSTEMP ON)
set(USE_ASYNC OFF)
set(USE_KLATT OFF)
set(USE_LIBPCAUDIO OFF)
set(USE_LIBSONIC OFF)
set(USE_MBROLA OFF)
set(USE_SPEECHPLAYER OFF)
add_subdirectory(${ESPEAK}/src espeak EXCLUDE_FROM_ALL)
option(OPENSPEECH_HOST_DATA_ONLY "Build only the data compiler" OFF)
if(NOT OPENSPEECH_HOST_DATA_ONLY)
add_library(openspeech-phonemes SHARED phonemes-jni.c)
target_link_libraries(openspeech-phonemes PRIVATE espeak-ng)
if(ANDROID)
    target_link_options(openspeech-phonemes PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(JNI REQUIRED)
    target_include_directories(openspeech-phonemes PRIVATE ${JNI_INCLUDE_DIRS})
endif()

endif()
